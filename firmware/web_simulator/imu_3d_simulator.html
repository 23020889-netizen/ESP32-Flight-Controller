Kết Nối Phần Cứng Thực Tế (WebSerial USB & Mã Nguồn ESP32 + MPU6050 / Xsens)

Cắm trực tiếp ESP32 hoặc bộ thu Xsens qua cổng COM (Trình duyệt Chrome / Edge) hoặc sao chép mã chuẩn bên dưới

Kết nối trực tiếp qua cổng USB Serial (Baudrate: 115200)

Định dạng chuỗi gửi từ ESP32 mỗi dòng: ax,ay,az,gx,gy,gz hoặc roll,pitch,yaw,ax,ay,az,gx,gy,gz

Mã giả / Firmware ESP32 đọc MPU6050 & Xsens chuẩn 100Hz: 
#include <Wire.h>
#include <MPU6050_tockn.h>

MPU6050 mpu6050(Wire);
unsigned long timer = 0;

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22); // SDA=21, SCL=22 trên ESP32
  mpu6050.begin();
  mpu6050.calcGyroOffsets(true); // Hiệu chuẩn Gyro khi khởi động
}

void loop() {
  mpu6050.update();
  // Gửi dữ liệu 100Hz (mỗi 10ms) lên Giao diện 3D Web
  if (millis() - timer >= 10) {
    timer = millis();
    Serial.print(mpu6050.getAngleX(), 2); Serial.print(","); // Roll
    Serial.print(mpu6050.getAngleY(), 2); Serial.print(","); // Pitch
    Serial.print(mpu6050.getAngleZ(), 2); Serial.print(","); // Yaw
    Serial.print(mpu6050.getAccX(), 3);   Serial.print(","); // Ax (g)
    Serial.print(mpu6050.getAccY(), 3);   Serial.print(","); // Ay (g)
    Serial.print(mpu6050.getAccZ(), 3);   Serial.print(","); // Az (g)
    Serial.print(mpu6050.getGyroX(), 2);  Serial.print(","); // Gx (deg/s)
    Serial.print(mpu6050.getGyroY(), 2);  Serial.print(","); // Gy (deg/s)
    Serial.println(mpu6050.getGyroZ(), 2);                   // Gz (deg/s)
  }
}
